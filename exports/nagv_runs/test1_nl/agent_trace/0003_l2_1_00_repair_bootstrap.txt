Every refund request is valid if and only if the refund request is initiated within 30 days of the original transaction date.

If a refund is a full refund, then the item associated with the refund is returned in the item's original, unopened packaging.

Every digital good is strictly non-refundable.

Every personalized item is strictly non-refundable.

Every approved refund is credited exclusively to the original payment method.

If the original payment method is inaccessible, the approved refund defaults to non-transferable store credit.

If it is not the case that the return is necessitated by a merchant error or a defect, then all original shipping charges are non-refundable.

If it is not the case that the return is necessitated by a merchant error or a defect, then all original handling charges are non-refundable.

# ---------------------------------------------------------------------------
# Units:
#   All monetary amounts are in cents (int).
#   days_since_transaction: int (number of days since original transaction).

@dataclass
class RefundRequest:
    # Timing
    days_since_transaction: int          # days since original transaction date

    # Item characteristics
    is_digital_good: bool
    is_personalized_item: bool

    original_shipping_cents: int         # original shipping charge
    original_handling_cents: int         # original handling charge

    # Authorization
    has_managerial_authorization: bool


@dataclass
class RefundDecision:
    is_valid_request: bool               # request passes validity gate
    is_refundable: bool                  # item is eligible for refund
    requires_original_packaging: bool    # must item be in original packaging?
    restocking_fee_cents: int            # restocking fee deducted (in cents)

    handling_refunded: bool              # True if handling is refunded

@Pure
def request_is_valid(r: RefundRequest) -> bool:
    # Valid iff initiated within 30 days AND accompanied by proof of purchase
    return (r.days_since_transaction >= 0 and
            r.days_since_transaction <= 30 and
            r.has_proof_of_purchase)


@Pure
def item_is_refundable(r: RefundRequest) -> bool:
    # Digital goods and personalized items are strictly non-refundable
    return not r.is_digital_good and not r.is_personalized_item


@Pure
def packaging_required(r: RefundRequest) -> bool:
    # Full refund => original unopened packaging required,
    # UNLESS item is verified defective on delivery.
    if r.is_full_refund and not r.is_defective_on_delivery:
        return True
    return False


@Pure
def restocking_fee(r: RefundRequest) -> int:
    # 15% restocking fee applies to opened, non-defective items.
    # 15% of item_price_cents, computed as integer (truncated).
    if r.is_opened and not r.is_defective_on_delivery:
        return (r.item_price_cents * 15) // 100
    return 0


@Pure
def return_due_to_merchant_or_defect(r: RefundRequest) -> bool:
    return r.is_merchant_error or r.is_defective_on_delivery


@Pure
def compute_refund_total(r: RefundRequest) -> int:
    # Base: item price minus restocking fee
    base = r.item_price_cents - restocking_fee(r)
    # Add shipping if refundable
    shipping = r.original_shipping_cents if return_due_to_merchant_or_defect(r) else 0
    # Add handling if refundable
    handling = r.original_handling_cents if return_due_to_merchant_or_defect(r) else 0
    return base + shipping + handling


@Pure
def authorization_required(refund_total_cents: int) -> bool:
    # Managerial authorization required if refund total exceeds $500 (50000 cents)
    return refund_total_cents > 50000

@Requires(r.original_shipping_cents >= 0)
@Requires(r.original_handling_cents >= 0)

@Ensures(Implies(Result().is_valid_request and Result().is_refundable and

    else:

        store_cred = False

        is_valid_request=valid,
        is_refundable=refundable,
        requires_original_packaging=pkg_required,
        restocking_fee_cents=fee,
        refund_total_cents=total,
        credited_to_original_method=credited,
        store_credit=store_cred,
        shipping_refunded=ship_refunded,

        authorization_satisfied=auth_satisfied,
