from nagini_contracts.contracts import Requires, Ensures, Pure, Implies, Result

#   days_since_transaction: int (number of days since original transaction).
# ---------------------------------------------------------------------------

@dataclass
class RefundRequest:
    # Timing
    days_since_transaction: int          # days since original transaction date

    # Item characteristics
    is_digital_good: bool
    is_personalized_item: bool
    is_defective_on_delivery: bool

    original_handling_cents: int         # original handling charge

@dataclass
class RefundDecision:
    is_valid_request: bool               # request passes validity gate
    is_refundable: bool                  # item is eligible for refund
    requires_original_packaging: bool    # must item be in original packaging?
    restocking_fee_cents: int            # restocking fee deducted (in cents)
    refund_total_cents: int              # final refund amount (in cents)
    credited_to_original_method: bool    # True => original method; False => store credit
    store_credit: bool                   # True if defaulting to store credit
    shipping_refunded: bool              # True if shipping is refunded
    handling_refunded: bool              # True if handling is refunded
    authorization_required: bool         # True if managerial auth required
    authorization_satisfied: bool        # True if auth requirement is met

# Pure helpers (called inside contracts)
# ---------------------------------------------------------------------------

@Pure
def request_is_valid(r: RefundRequest) -> bool:
    return (r.days_since_transaction >= 0 and
            r.days_since_transaction <= 30 and
            r.has_proof_of_purchase)


@Pure
def item_is_refundable(r: RefundRequest) -> bool:
    return not r.is_digital_good and not r.is_personalized_item


@Pure
def packaging_required(r: RefundRequest) -> bool:
    if r.is_full_refund and not r.is_defective_on_delivery:
        return True
    return False


@Pure
def restocking_fee(r: RefundRequest) -> int:
    if r.is_opened and not r.is_defective_on_delivery:
        return (r.item_price_cents * 15) // 100
    return 0


@Pure
def return_due_to_merchant_or_defect(r: RefundRequest) -> bool:
    return r.is_merchant_error or r.is_defective_on_delivery


@Pure
def compute_refund_total(r: RefundRequest) -> int:
    base = r.item_price_cents - restocking_fee(r)
    shipping = r.original_shipping_cents if return_due_to_merchant_or_defect(r) else 0
    handling = r.original_handling_cents if return_due_to_merchant_or_defect(r) else 0
    return base + shipping + handling


@Pure
def authorization_required(refund_total_cents: int) -> bool:
    return refund_total_cents > 50000


@Pure
def decision_valid_and_refundable(d: RefundDecision) -> bool:
    return d.is_valid_request and d.is_refundable


@Pure
def decision_auth_required(d: RefundDecision) -> bool:
    return d.authorization_required

# Main policy function
# ---------------------------------------------------------------------------

@Requires(r.item_price_cents >= 0)  # type: ignore[name-defined]
@Requires(r.original_shipping_cents >= 0)  # type: ignore[name-defined]
@Requires(r.original_handling_cents >= 0)  # type: ignore[name-defined]

@Ensures(Implies(not item_is_refundable(r), not Result().is_refundable))  # type: ignore[name-defined]

@Ensures(Implies(not decision_valid_and_refundable(Result()),

@Ensures(Result().shipping_refunded == return_due_to_merchant_or_defect(r))  # type: ignore[name-defined]
@Ensures(Result().handling_refunded == return_due_to_merchant_or_defect(r))  # type: ignore[name-defined]
@Ensures(Result().authorization_required ==

@Ensures(Implies(not decision_auth_required(Result()),

@Ensures(Implies(decision_valid_and_refundable(Result()) and

def evaluate_refund(r: RefundRequest) -> RefundDecision:
    valid = request_is_valid(r)
    refundable = item_is_refundable(r)

    pkg_required = packaging_required(r)
    fee = restocking_fee(r)

    if valid and refundable:
        total = compute_refund_total(r)
    else:
        total = 0

    ship_refunded = return_due_to_merchant_or_defect(r)
    hand_refunded = return_due_to_merchant_or_defect(r)

    auth_req = authorization_required(compute_refund_total(r)) and valid and refundable
    auth_satisfied = (not auth_req) or r.has_managerial_authorization

    if valid and refundable:
        if r.original_payment_method_accessible:
            credited = True
            store_cred = False
        else:
            credited = False
            store_cred = True
    else:
        credited = False
        store_cred = False

    return RefundDecision(
        is_valid_request=valid,
        is_refundable=refundable,
        requires_original_packaging=pkg_required,
        restocking_fee_cents=fee,
        refund_total_cents=total,
        credited_to_original_method=credited,
        store_credit=store_cred,
        shipping_refunded=ship_refunded,
        handling_refunded=hand_refunded,
        authorization_required=auth_req,
        authorization_satisfied=auth_satisfied,
    )
