The Nagini errors indicate "insufficient permission to access" fields of `req` inside `@Pure` function contracts. In Nagini, when a `@Pure` function takes a reference-typed parameter and accesses its fields in `Requires`/`Ensures`, we need to grant read permission via `Acc(...)`. The fix is to add `Requires(Acc(req.field, ...))` for each accessed field, or more practically, use a single predicate or list all `Acc` requires.

The standard pattern for Nagini with `@Pure` functions accessing object fields: include `Requires(Acc(req.field))` for each field read. Since these are simple read-only pure functions, we need read permissions.

@Pure
def is_within_window(req: RefundRequest) -> bool:
    Requires(Acc(req.days_since_transaction))
    Ensures(Result() == (req.days_since_transaction <= 30))
    return req.days_since_transaction <= 30


@Pure
def is_request_valid(req: RefundRequest) -> bool:
    # Line 0: validity iff within 30 days.
    Requires(Acc(req.days_since_transaction))
    Ensures(Result() == is_within_window(req))
    return is_within_window(req)

def packaging_requirement_satisfied(req: RefundRequest) -> bool:

    Requires(Acc(req.is_full_refund))
    Requires(Acc(req.verified_defective))

    Ensures(Result() == ((not req.is_full_refund) or req.verified_defective or req.returned_in_original_packaging))

    # packaging requirement satisfied, and managerial authorization for >$500.
    Requires(Acc(req.days_since_transaction))
    Requires(Acc(req.has_proof_of_purchase))
    Requires(Acc(req.is_digital_good))
    Requires(Acc(req.is_personalized))
    Requires(Acc(req.is_full_refund))

@Pure
def compute_refund_amount(req: RefundRequest) -> int:
    # Refund total after restocking fee (15% deduction if applicable).
    Requires(Acc(req.is_opened))
    Requires(Acc(req.verified_defective))
    Requires(Acc(req.refund_total))
    Ensures(Implies(restocking_fee_applies(req), Result() == req.refund_total - (req.refund_total * 15) // 100))
    Ensures(Implies(not restocking_fee_applies(req), Result() == req.refund_total))
    if restocking_fee_applies(req):
        return req.refund_total - (req.refund_total * 15) // 100
